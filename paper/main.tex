\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig1_regret_comparison.pdf}
  \caption{
    Regret comparison over $T=500$ search steps (50 independent runs, shaded = $\pm$1 SD).
    \textbf{(a)} Cumulative regret: DVA-MCTS achieves substantially lower cumulative
    regret than Uniform Verification despite fewer verifier calls.
    \textbf{(b)} Log-log scaling confirms the $O(\sqrt{T})$ rate for DVA-MCTS
    (slope $\approx 0.5$), matching Theorem~\ref{thm:main}.
  }
  \label{fig:regret}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig2_verifier_calls.pdf}
  \caption{
    Verifier call efficiency as a function of search budget $B$.
    \textbf{(a)} Absolute verifier calls: DVA-MCTS grows as $O(\sqrt{B}\log B)$
    versus $O(B)$ for Uniform Verification.
    \textbf{(b)} DVA-MCTS calls as a percentage of Uniform Verification calls,
    confirming the increasing efficiency advantage at larger budgets (Theorem~\ref{thm:calls}).
  }
  \label{fig:calls}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig3_lipschitz.pdf}
  \caption{
    Empirical validation of the Lipschitz score continuity assumption
    (Assumption~\ref{ass:lipschitz}) on simulated PRM trajectories.
    \textbf{(a)} Score differences $|V(s_d) - V(s_{d'})|$ vs. depth gap
    $|d-d'|$ with the Lipschitz envelope $L=0.35$.
    \textbf{(b)} Per-bin estimates of $\hat{L}$ are stable across depth gaps
    (CV$< 0.1$).
  }
  \label{fig:lipschitz}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.65\linewidth]{figures/fig4_compute_accuracy.pdf}
  \caption{
    Compute--accuracy tradeoff. DVA-MCTS tracks Oracle Allocation closely,
    reaching within $2.1\%$ accuracy at $T=800$ while using $38\%$ fewer
    verifier calls. Uniform Verification and Random Allocation are shown for
    reference.
  }
  \label{fig:tradeoff}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig5_ablation.pdf}
  \caption{
    Ablation over threshold exponent $\alpha$ (where $\tau_t = \gamma/t^\alpha$).
    \textbf{(a)} Cumulative regret for each $\alpha$; $\alpha=0.5$ matches the
    $O(\sqrt{T})$ reference most closely.
    \textbf{(b)} Operating curve: regret at $T=400$ versus verification rate.
    The Pareto-optimal point is near $\alpha=0.5$, consistent with the theory.
  }
  \label{fig:ablation}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig6_estimated_L.pdf}
  \caption{
    Effect of using the estimated Lipschitz constant $\hat{L}{=}0.091$ instead
    of the true $L{=}0.35$ ($T{=}400$, 50 runs).
    \textbf{(a)} Cumulative regret curves; the underestimated $\hat{L}$ yields
    $43\%$ higher final regret.
    \textbf{(b)} Final regret and verifier call count side by side.
    Using $\hat{L}$ saves $8\%$ of verifier calls at the cost of $43\%$ more
    regret---a meaningful but non-catastrophic degradation (see
    Section~\ref{sec:lhat} and Remark~\ref{rem:lhat}).
  }
  \label{fig:lhat}
\end{figure}
