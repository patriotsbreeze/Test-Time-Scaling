\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig1_regret_comparison.pdf}
  \caption{
    Regret comparison over $T{=}400$ search steps (50 matched-pair runs,
    shaded $= \pm1$ SD).
    \textbf{(a)} Cumulative regret: no algorithm dominates significantly in the
    exploration regime ($T \ll K^D$; Wilcoxon $p{=}0.109$ for DVA vs.\ Uniform).
    DVA achieves $98\%$ accuracy, Uniform and Random achieve $100\%$
    (Table~\ref{tab:summary}).
    \textbf{(b)} Log-log scaling: empirical slope $0.635$ ($R^2{=}0.961$), above
    the asymptotic $0.5$ prediction due to exploration-regime overhead
    (Remark~\ref{rem:exploration}).
  }
  \label{fig:regret}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig2_verifier_calls.pdf}
  \caption{
    Verifier call efficiency as a function of search budget $B$.
    \textbf{(a)} Absolute verifier calls: DVA-MCTS is bounded by $N_{\mathcal{T}}{=}K(K^D{-}1)/(K{-}1)$
    (a constant), saturating well below $O(B)$ for Uniform Verification.
    In the transition regime ($B \leq N_{\mathcal{T}}$), empirical counts grow as
    ${\approx}1.9\sqrt{B}\log B$.
    \textbf{(b)} DVA-MCTS calls as a percentage of Uniform Verification calls,
    confirming the increasing efficiency advantage at larger budgets (Theorem~\ref{thm:calls}).
  }
  \label{fig:calls}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig3_lipschitz.pdf}
  \caption{
    Empirical validation of the Lipschitz score continuity assumption
    (Assumption~\ref{ass:lipschitz}) on simulated PRM trajectories.
    \textbf{(a)} Score differences $|V(s_d) - V(s_{d'})|$ vs. depth gap
    $|d-d'|$ with the Lipschitz envelope $L=0.35$.
    \textbf{(b)} Per-bin estimates of $\hat{L}$ are stable across depth gaps
    (CV$< 0.1$).
  }
  \label{fig:lipschitz}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=0.65\linewidth]{figures/fig4_compute_accuracy.pdf}
  \caption{
    Compute--accuracy tradeoff (50 runs per budget).
    DVA-MCTS matches Uniform Verification within $2$ percentage points at all
    budgets $B \geq 400$ while using up to $73\%$ fewer verifier calls at
    $B{=}3{,}200$. Random-path Best-of-$N$ fails in deep trees
    ($0\%$ accuracy; see Section~\ref{sec:tradeoff_results}).
  }
  \label{fig:tradeoff}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig5_ablation.pdf}
  \caption{
    Ablation over threshold exponent $\alpha$ (where $\tau_t = \gamma/t^\alpha$).
    \textbf{(a)} Cumulative regret for each $\alpha$; $\alpha=0.5$ matches the
    $O(\sqrt{T})$ reference most closely.
    \textbf{(b)} Operating curve: regret at $T=400$ versus verification rate.
    The Pareto-optimal point is near $\alpha=0.5$, consistent with the theory.
  }
  \label{fig:ablation}
\end{figure}

\begin{figure}[t]
  \centering
  \includegraphics[width=\linewidth]{figures/fig6_estimated_L.pdf}
  \caption{
    Effect of Lipschitz constant specification ($T{=}400$, 50 matched-pair runs;
    Table~\ref{tab:lhat}).
    \textbf{(a)} Grouped bar chart of mean final regret; in the exploration regime,
    all three configurations perform comparably (no pairwise Wilcoxon test significant
    at $\alpha{=}0.05$).  The adaptive estimator achieves the lowest regret ($23.88$)
    and $100\%$ accuracy; see Section~\ref{sec:lhat} and Remark~\ref{rem:lhat}.
    \textbf{(b)} Regret--calls scatter: adaptive $\hat{L}$ Pareto-dominates
    fixed $\hat{L}$ (lower regret, more calls); both outperform oracle $L$ in this
    exploration regime.
  }
  \label{fig:lhat}
\end{figure}
